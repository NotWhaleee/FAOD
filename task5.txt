; 5.1

(defn get-blocks [coll n]
  (loop [coll coll
         res []]
    (if (empty? coll)
      res
      (recur (drop n coll) (conj res (take n coll))))))

(defn parallel-filter [pred coll block-size]
  (let [blocks (get-blocks coll block-size)
        futs (doall (map (fn [b] (future (doall (filter pred b)))) blocks))]
    (doall (apply concat (map deref futs)))))

; 5.2

(defn lazy-blocks [coll n]
  (lazy-seq
    (when (not (empty? coll))
      (cons (take n coll) (lazy-blocks (drop n coll) n)))))

; ahead is futs moved forward by k, so at most k + 1 futures are running
(defn lazy-step [futs ahead]
  (lazy-seq
    (when (not (empty? futs))
      (seq ahead)
      (concat (deref (first futs))
              (lazy-step (rest futs) (rest ahead))))))

(defn lazy-parallel-filter [pred coll block-size]
  (let [k 4
        futs (map (fn [b] (future (doall (filter pred b))))
                  (lazy-blocks coll block-size))]
    (lazy-step futs (drop k futs))))


; slow predicate for tests
(defn prime? [n]
  (if (< n 2)
    false
    (loop [d 2]
      (cond
        (> (* d d) n) true
        (= 0 (mod n d)) false
        :else (recur (inc d))))))

(defn slow-prime? [n]
  (prime? (+ 1000000000 n)))
